function convertToColoringBook() { if (!originalImageObj) { alert("الرجاء رفع صورة ملونة أولاً!"); return; } loadingOverlay.style.display = 'flex'; // نستخدم Canvas مؤقت للمعالجة const tempCanvas = document.createElement('canvas'); tempCanvas.width = canvas.width; tempCanvas.height = canvas.height; const tCtx = tempCanvas.getContext('2d'); // 1. تنظيف وتنعيم الصورة (Blur) لإزالة الضجيج غير المرغوب فيه tCtx.filter = 'blur(1.5px) grayscale(100%) brightness(110%)'; tCtx.drawImage(originalImageObj, 0, 0, canvas.width, canvas.height); // 2. تطبيق تأثير التباين العالي (High Contrast) لإظهار الخطوط الرئيسية فقط tCtx.filter = 'contrast(200%)'; tCtx.drawImage(tempCanvas, 0, 0); // 3. تحويل الألوان بحيث تصبح الخلفية بيضاء والخطوط سوداء واضحة const imgData = tCtx.getImageData(0, 0, canvas.width, canvas.height); const data = imgData.data; for (let i = 0; i < data.length; i += 4) { // حد العتبة (Threshold) لجعل الخطوط حادة جداً let v = data[i] > 180 ? 255 : 0; data[i] = data[i + 1] = data[i + 2] = v; } tCtx.putImageData(imgData, 0, 0); // عرض النتيجة ctx.fillStyle = "#ffffff"; ctx.fillRect(0, 0, canvas.width, canvas.height); ctx.drawImage(tempCanvas, 0, 0); currentModeObj = new Image(); currentModeObj.src = tempCanvas.toDataURL(); saveState(); loadingOverlay.style.display = 'none'; }
