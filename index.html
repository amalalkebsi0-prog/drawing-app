// خوارزمية تحويل الصورة إلى رسم خطي نظيف وجاهز للتلوين document.getElementById('btnOutline').onclick = () => { if (!originalImage) { alert('يرجى رفع صورة أولاً!'); return; } const width = canvas.width; const height = canvas.height; // 1. رسم الصورة على الكانفاس واستخراج بيانات البكسلات ctx.drawImage(originalImage, 0, 0, width, height); const imgData = ctx.getImageData(0, 0, width, height); const data = imgData.data; // 2. تحويل الصورة للتدرج الرمادي (Grayscale) const gray = new Float32Array(width * height); for (let i = 0; i < data.length; i += 4) { gray[i / 4] = 0.299 * data[i] + 0.587 * data[i + 1] + 0.114 * data[i + 2]; } // 3. تنعيم الصورة لمسح الضوضاء والظلال الخفيفة (Box Blur) const blurred = new Float32Array(width * height); const radius = 2; // نطاق التنعيم for (let y = 0; y < height; y++) { for (let x = 0; x < width; x++) { let sum = 0; let count = 0; for (let dy = -radius; dy <= radius; dy++) { for (let dx = -radius; dx <= radius; dx++) { const nx = x + dx; const ny = y + dy; if (nx >= 0 && nx < width && ny >= 0 && ny < height) { sum += gray[ny * width + nx]; count++; } } } blurred[y * width + x] = sum / count; } } // 4. مقارنة البكسل الأصلي بالبكسل المُنعم لاستخراج الخطوط الخارجية فقط (Difference Edge Detection) const output = ctx.createImageData(width, height); const outData = output.data; for (let i = 0; i < gray.length; i++) { const diff = blurred[i] - gray[i]; // رسم خط أسود رفيع فقط عند وجود حافة واضحة، وتفريغ الباقي بالأبيض const val = diff > 3.5 ? 0 : 255; const idx = i * 4; outData[idx] = val; // R outData[idx + 1] = val; // G outData[idx + 2] = val; // B outData[idx + 3] = 255; // Alpha } ctx.putImageData(output, 0, 0); saveState(); };
