document.addEventListener("DOMContentLoaded", () => { const canvas = document.getElementById("paintCanvas"); const ctx = canvas.getContext("2d", { willReadFrequently: true }); // عناصر التحكم const imageLoader = document.getElementById("imageLoader"); const btnOutline = document.getElementById("btnOutline"); const btnUndo = document.getElementById("btnUndo"); const btnClear = document.getElementById("btnClear"); const btnSave = document.getElementById("btnSave"); const lineWidthInput = document.getElementById("lineWidth"); const widthValue = document.getElementById("widthValue"); const customColorInput = document.getElementById("customColor"); const fillShapeCheckbox = document.getElementById("fillShape"); const referenceWrapper = document.getElementById("referenceWrapper"); const referenceImage = document.getElementById("referenceImage"); // المتغيرات العامة let isDrawing = false; let currentTool = "brush"; let currentColor = "#000000"; let currentWidth = 5; let selectedEmoji = "⭐"; let startX = 0; let startY = 0; let snapshot = null; // سجّل التراجع (Undo History) const historyStack = []; const MAX_HISTORY = 20; // تهيئة الكانفاس ببلون أبيض ctx.fillStyle = "#ffffff"; ctx.fillRect(0, 0, canvas.width, canvas.height); saveState(); // --- حفظ حالة الكانفاس للتراجع --- function saveState() { if (historyStack.length >= MAX_HISTORY) historyStack.shift(); historyStack.push(ctx.getImageData(0, 0, canvas.width, canvas.height)); } // --- تراجع عن الخطوة الأخيرة --- btnUndo.addEventListener("click", () => { if (historyStack.length > 1) { historyStack.pop(); // حذف الحالة الحالية const previousState = historyStack[historyStack.length - 1]; ctx.putImageData(previousState, 0, 0); } }); // --- مسح الكانفاس --- btnClear.addEventListener("click", () => { ctx.fillStyle = "#ffffff"; ctx.fillRect(0, 0, canvas.width, canvas.height); saveState(); }); // --- حفظ الصورة --- btnSave.addEventListener("click", () => { const link = document.createElement("a"); link.download = "my-drawing.png"; link.href = canvas.toDataURL(); link.click(); }); // --- اختيار الألوان --- document.querySelectorAll(".color-dot").forEach((dot) => { dot.addEventListener("click", (e) => { document.querySelector(".color-dot.active")?.classList.remove("active"); e.target.classList.add("active"); currentColor = e.target.getAttribute("data-color"); customColorInput.value = currentColor; }); }); customColorInput.addEventListener("input", (e) => { currentColor = e.target.value; }); // --- سماكة الفرشاة --- lineWidthInput.addEventListener("input", (e) => { currentWidth = e.target.value; widthValue.textContent = `${currentWidth} px`; }); // --- اختيار الأدوات والأشكال --- document.querySelectorAll(".tool-btn").forEach((btn) => { btn.addEventListener("click", (e) => { document.querySelectorAll(".tool-btn").forEach((b) => b.classList.remove("active")); const targetBtn = e.target.closest(".tool-btn"); targetBtn.classList.add("active"); currentTool = targetBtn.getAttribute("data-tool"); }); }); // --- اختيار الملصقات --- document.querySelectorAll(".emoji-btn").forEach((btn) => { btn.addEventListener("click", (e) => { selectedEmoji = e.target.getAttribute("data-emoji"); currentTool = "sticker"; document.querySelectorAll(".tool-btn").forEach((b) => b.classList.remove("active")); }); }); // --- أحداث الرسم بالماوس واللمس --- function getCoordinates(e) { const rect = canvas.getBoundingClientRect(); const clientX = e.touches ? e.touches[0].clientX : e.clientX; const clientY = e.touches ? e.touches[0].clientY : e.clientY; return { x: clientX - rect.left, y: clientY - rect.top }; } function startDraw(e) { isDrawing = true; const { x, y } = getCoordinates(e); startX = x; startY = y; ctx.beginPath(); ctx.moveTo(startX, startY); snapshot = ctx.getImageData(0, 0, canvas.width, canvas.height); if (currentTool === "fill") { floodFill(Math.floor(startX), Math.floor(startY), currentColor); saveState(); isDrawing = false; } else if (currentTool === "sticker") { ctx.font = `${currentWidth * 3}px sans-serif`; ctx.textAlign = "center"; ctx.textBaseline = "middle"; ctx.fillText(selectedEmoji, startX, startY); saveState(); isDrawing = false; } } function drawing(e) { if (!isDrawing) return; const { x, y } = getCoordinates(e); ctx.strokeStyle = currentTool === "eraser" ? "#ffffff" : currentColor; ctx.fillStyle = currentColor; ctx.lineWidth = currentWidth; ctx.lineCap = "round"; ctx.lineJoin = "round"; if (["brush", "pen", "eraser"].includes(currentTool)) { ctx.lineTo(x, y); ctx.stroke(); } else if (currentTool === "spray") { for (let i = 0; i < 20; i++) { const offsetX = (Math.random() - 0.5) * currentWidth * 2; const offsetY = (Math.random() - 0.5) * currentWidth * 2; ctx.fillRect(x + offsetX, y + offsetY, 1, 1); } } else { // رسم الأشكال الهندسية مع معاينة حية ctx.putImageData(snapshot, 0, 0); drawShape(x, y); } } function stopDraw() { if (isDrawing) { isDrawing = false; saveState(); } } // --- رسم الأشكال الهندسية --- function drawShape(x, y) { const isFilled = fillShapeCheckbox.checked; ctx.beginPath(); if (currentTool === "rectangle") { const w = x - startX; const h = y - startY; if (isFilled) ctx.fillRect(startX, startY, w, h); ctx.strokeRect(startX, startY, w, h); } else if (currentTool === "circle") { const radius = Math.sqrt(Math.pow(x - startX, 2) + Math.pow(y - startY, 2)); ctx.arc(startX, startY, radius, 0, 2 * Math.PI); if (isFilled) ctx.fill(); ctx.stroke(); } else if (currentTool === "line") { ctx.moveTo(startX, startY); ctx.lineTo(x, y); ctx.stroke(); } else if (currentTool === "triangle") { ctx.moveTo(startX, startY); ctx.lineTo(x, y); ctx.lineTo(startX * 2 - x, y); ctx.closePath(); if (isFilled) ctx.fill(); ctx.stroke(); } } // --- خوارزمية دلو التعبئة (Flood Fill) --- function floodFill(startX, startY, fillColorHex) { const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height); const data = imgData.data; const targetColor = getPixelColor(data, startX, startY); const fillColor = hexToRgba(fillColorHex); if (colorsMatch(targetColor, fillColor)) return; const pixelStack = [[startX, startY]]; const width = canvas.width; const height = canvas.height; while (pixelStack.length) { const [x, y] = pixelStack.pop(); let currentY = y; while (currentY >= 0 && matchTargetColor(data, x, currentY, width, targetColor)) { currentY--; } currentY++; let reachLeft = false; let reachRight = false; while (currentY < height && matchTargetColor(data, x, currentY, width, targetColor)) { setPixelColor(data, x, currentY, width, fillColor); if (x > 0) { if (matchTargetColor(data, x - 1, currentY, width, targetColor)) { if (!reachLeft) { pixelStack.push([x - 1, currentY]); reachLeft = true; } } else if (reachLeft) reachLeft = false; } if (x < width - 1) { if (matchTargetColor(data, x + 1, currentY, width, targetColor)) { if (!reachRight) { pixelStack.push([x + 1, currentY]); reachRight = true; } } else if (reachRight) reachRight = false; } currentY++; } } ctx.putImageData(imgData, 0, 0); } function getPixelColor(data, x, y) { const idx = (y * canvas.width + x) * 4; return [data[idx], data[idx + 1], data[idx + 2], data[idx + 3]]; } function setPixelColor(data, x, y, width, color) { const idx = (y * width + x) * 4; data[idx] = color[0]; data[idx + 1] = color[1]; data[idx + 2] = color[2]; data[idx + 3] = 255; } function matchTargetColor(data, x, y, width, target) { const color = getPixelColor(data, x, y); return Math.abs(color[0] - target[0]) < 30 && Math.abs(color[1] - target[1]) < 30 && Math.abs(color[2] - target[2]) < 30; } function colorsMatch(a, b) { return a[0] === b[0] && a[1] === b[1] && a[2] === b[2]; } function hexToRgba(hex) { let c = hex.replace("#", ""); if (c.length === 3) c = c.split("").map((x) => x + x).join(""); const num = parseInt(c, 16); return [(num >> 16) & 255, (num >> 8) & 255, num & 255, 255]; } // --- رفع الصور وتفريغ الألوان (تطبيق التلوين) --- imageLoader.addEventListener("change", (e) => { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = (evt) => { const img = new Image(); img.onload = () => { ctx.fillStyle = "#ffffff"; ctx.fillRect(0, 0, canvas.width, canvas.height); ctx.drawImage(img, 0, 0, canvas.width, canvas.height); referenceImage.src = evt.target.result; referenceWrapper.style.display = "block"; saveState(); }; img.src = evt.target.result; }; reader.readAsDataURL(file); }); // تفريغ الألوان لتصبح الصورة للرسم والتلوين (Black & White Outline) btnOutline.addEventListener("click", () => { const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height); const d = imgData.data; // تحويل إلى رمادي كخطوة أولى for (let i = 0; i < d.length; i += 4) { const avg = (d[i] + d[i + 1] + d[i + 2]) / 3; // تطبيق حد سواد عالي للتحويل إلى أبيض وأسود صريح const val = avg < 150 ? 0 : 255; d[i] = val; d[i + 1] = val; d[i + 2] = val; } ctx.putImageData(imgData, 0, 0); saveState(); }); // الربط بالأحداث canvas.addEventListener("mousedown", startDraw); canvas.addEventListener("mousemove", drawing); canvas.addEventListener("mouseup", stopDraw); canvas.addEventListener("mouseleave", stopDraw); // دعم الأجهزة اللوحية والمحمول canvas.addEventListener("touchstart", (e) => { e.preventDefault(); startDraw(e); }); canvas.addEventListener("touchmove", (e) => { e.preventDefault(); drawing(e); }); canvas.addEventListener("touchend", stopDraw); });
