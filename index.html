document.getElementById('btnOutline').onclick = () => { if (!originalImage) { alert('يرجى رفع صورة أولاً!'); // "Please upload an image first!" return; } const width = canvas.width; const height = canvas.height; // 1. Draw image to canvas and extract raw pixel data ctx.drawImage(originalImage, 0, 0, width, height); const imgData = ctx.getImageData(0, 0, width, height); const data = imgData.data; // 2. Convert to Grayscale const gray = new Float32Array(width * height); for (let i = 0; i < data.length; i += 4) { gray[i / 4] = 0.299 * data[i] + 0.587 * data[i + 1] + 0.114 * data[i + 2]; } // 3. Smooth image (Box Blur) to remove noise and soft shadows const blurred = new Float32Array(width * height); const radius = 2; // Blur radius for (let y = 0; y < height; y++) { for (let x = 0; x < width; x++) { let sum = 0; let count = 0; for (let dy = -radius; dy <= radius; dy++) { for (let dx = -radius; dx <= radius; dx++) { const nx = x + dx; const ny = y + dy; if (nx >= 0 && nx < width && ny >= 0 && ny < height) { sum += gray[ny * width + nx]; count++; } } } blurred[y * width + x] = sum / count; } } // 4. Difference Edge Detection (extract thin outer outlines) const output = ctx.createImageData(width, height); const outData = output.data; for (let i = 0; i < gray.length; i++) { const diff = blurred[i] - gray[i]; // Draw black line if sharp edge detected (diff > 3.5), otherwise white const val = diff > 3.5 ? 0 : 255; const idx = i * 4; outData[idx] = val; // Red outData[idx + 1] = val; // Green outData[idx + 2] = val; // Blue outData[idx + 3] = 255; // Alpha } ctx.putImageData(output, 0, 0); saveState(); };
